    SET v_jam = SEC_TO_TIME(32400 + ((i * 413) MOD 46800));
    SET v_penerima = ELT(
      1 + (i MOD 12),
      'Yuandi', 'Yuruf', 'Rufiana', 'Wahyuari',
      'Bagus Wijaya', 'Sari Dewi', 'Kadek Adi', 'Luh Putu',
      'Made Putra', 'Komang Ayu', 'Yuliana', 'Sandy Pratama'
    );
    SET v_alamat = ELT(

      'Denpasar Utara',
      'Gianyar',
      'Badung',
      'Tabanan',

    );
    SET v_tbayar = 45000 + ((i * 19750) MOD 550000);
    SET v_kasir = ELT(1 + (i MOD 4), 'yulia', 'sandy', 'rina', 'doni');
    SET v_diskon = CASE WHEN i MOD 3 = 0 THEN 10000 WHEN i MOD 2 = 0 THEN 5000 ELSE 0 END;

    INSERT INTO `invoices` (`idinvoice`, `tgl`, `jam`, `penerima`, `alamat`, `tbayar`, `kasir`, `diskon`)

DROP TABLE IF EXISTS `invoicedatail`;

CREATE TABLE `invoicedatail` (
  `iddetail` int(11) NOT NULL AUTO_INCREMENT,
  `idinvoice` int(11) NOT NULL,
  `idbarang` varchar(5) NOT NULL,
  `qty` int(11) NOT NULL,
  `diskon` float DEFAULT NULL,
  PRIMARY KEY (`iddetail`),
  KEY `fk_detail_invoice` (`idinvoice`),
  KEY `fk_detail_stok` (`idbarang`),
  CONSTRAINT `fk_detail_invoice` FOREIGN KEY (`idinvoice`) REFERENCES `invoices` (`idinvoice`) ON UPDATE CASCADE,
  CONSTRAINT `fk_detail_stok` FOREIGN KEY (`idbarang`) REFERENCES `stokbarang` (`idbarang`) ON UPDATE CASCADE
) ENGINE=InnoDB AUTO_INCREMENT=9 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

DROP PROCEDURE IF EXISTS `seed_invoicedatail`;
DELIMITER $$
CREATE PROCEDURE `seed_invoicedatail`()
BEGIN
  DECLARE inv INT DEFAULT 1;
  DECLARE idx INT;
  DECLARE brg_num INT;
  DECLARE v_idbarang varchar(5);
  DECLARE v_qty int(11);
  DECLARE v_diskon float;

  WHILE inv <= 100 DO
    SET idx = 0;

    WHILE idx < 6 DO
      SET brg_num = 1 + (((inv * 7) + (idx * 9)) MOD 50);
      SET v_idbarang = CONCAT('B', LPAD(brg_num, 4, '0'));

      IF NOT EXISTS (
        SELECT 1 FROM `invoicedatail`
        WHERE `idinvoice` = inv AND `idbarang` = v_idbarang
      ) THEN
        SET v_qty = 1 + ((inv + idx) MOD 4);
        SET v_diskon = CASE WHEN (inv + idx) MOD 6 = 0 THEN 5000 ELSE 0 END;

        INSERT INTO `invoicedatail` (`idinvoice`, `idbarang`, `qty`, `diskon`)
        VALUES (inv, v_idbarang, v_qty, v_diskon);
      END IF;

      SET idx = idx + 1;
    END WHILE;

    SET inv = inv + 1;
  END WHILE;
END$$
DELIMITER ;

CALL `seed_invoicedatail`();
DROP PROCEDURE IF EXISTS `seed_invoicedatail`;
